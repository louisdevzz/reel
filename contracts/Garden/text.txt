    // ===================== Pot Functions =====================
    public entry fun create_pot(
        name: String,
        description: String,
        uri: String,
        receive_addr: address
    ) acquires PotCollectionOwnerConfig {
        let pot_collection_address = get_pot_collection_owner_address();
        let constructor_ref = &token::create(
            &get_pot_collection_signer(pot_collection_address),
            name,
            description,
            name,
            option::none(),
            uri,
        );
        let token_signer_ref = &object::generate_signer(constructor_ref);
        let pot_address = signer::address_of(token_signer_ref);

        let extend_ref = object::generate_extend_ref(constructor_ref);
        let burn_ref = token::generate_burn_ref(constructor_ref);
        let transfer_ref = object::generate_transfer_ref(constructor_ref);

        let pot = Pot {
            name: name,
            level: 1,
            slot_count: 1,
            current_plants: vector::empty<address>(),
            quantity: 1,
            extend_ref,
            burn_ref,
        };
        move_to(token_signer_ref, pot);

        // Emit event for minting Pot
        event::emit<MintPotEvent>(MintPotEvent {
            pot_address,
            name,
            description,
            uri,
            receiver: receive_addr,
        });

        // Transfer the Pot to the receive_addr
        object::transfer_with_ref(object::generate_linear_transfer_ref(&transfer_ref), receive_addr);
    }

    public entry fun plant_seed(pot_addr: address, plant_addr: address) acquires Pot, Plant {
        assert!(exists<Pot>(pot_addr), 0x60007); // Pot does not exist
        let pot = borrow_global_mut<Pot>(pot_addr);
        assert!(exists<Plant>(plant_addr), 0x60008); // Plant does not exist
        let plant = borrow_global_mut<Plant>(plant_addr);
        let current_count = vector::length(&pot.current_plants);
        assert!(current_count < (pot.slot_count as u64), 1001); // Error: Pot is full
        
        let now = timestamp::now_seconds();
        let base_duration = plant.base_grow_duration_sec;
        let bonus = get_growth_bonus(pot.level);
        let effective_duration = base_duration * (100 - bonus) / 100;
        
        plant.pot_address = option::some(pot_addr);
        plant.planted_at = now;
        plant.growth_complete_at = now + effective_duration;
        plant.extended_growth_percent = bonus as u8;
        plant.stage = 0;
        vector::push_back(&mut pot.current_plants, plant_addr);
    }

    public entry fun update_growth_stage(plant_addr: address) acquires Plant {
        assert!(exists<Plant>(plant_addr), 0x60008); // Plant does not exist
        let plant = borrow_global_mut<Plant>(plant_addr);
        
        let now = timestamp::now_seconds();
        let elapsed = now - plant.planted_at;
        let total = plant.growth_complete_at - plant.planted_at;
        
        let percent = 100 * elapsed / total;
        
        if (percent >= 100) {
            plant.stage = 3;
            plant.is_ready_to_harvest = true;
        } else if (percent >= 70) {
            plant.stage = 2;
        } else if (percent >= 30) {
            plant.stage = 1;
        } else {
            plant.stage = 0;
        }
    }

    public entry fun upgrade_pot(pot_addr: address) acquires Pot {
        assert!(exists<Pot>(pot_addr), 0x60007); // Pot does not exist
        let pot = borrow_global_mut<Pot>(pot_addr);
        assert!(pot.level < 5, 1002); // Error: Pot already at max level
        
        pot.level = pot.level + 1;
        pot.slot_count = pot.slot_count + 1;
    }

    // Recursive helper for harvest_all
    fun harvest_all_helper(plants: &mut vector<address>) acquires Plant {
        if (vector::is_empty(plants)) {
            return;
        };
        let plant_addr = *vector::borrow(plants, 0);
        assert!(exists<Plant>(plant_addr), 0x60008); // Plant does not exist
        let plant = borrow_global_mut<Plant>(plant_addr);
        if (plant.is_ready_to_harvest) {
            let Plant {
                plant_id: _,
                species: _,
                stage: _,
                growth: _,
                planted_at: _,
                base_grow_duration_sec: _,
                extended_growth_percent: _,
                growth_complete_at: _,
                is_ready_to_harvest: _,
                pot_address: _,
                quantity: _,
                extend_ref: _,
                burn_ref,
            } = move_from<Plant>(plant_addr);
            token::burn(burn_ref);
            vector::remove(plants, 0);
            harvest_all_helper(plants);
        } else {
            return;
        }
    }

    public entry fun harvest_all(pot_addr: address) acquires Pot, Plant {
        assert!(exists<Pot>(pot_addr), 0x60007); // Pot does not exist
        let pot = borrow_global_mut<Pot>(pot_addr);
        let plants = &mut pot.current_plants;
        harvest_all_helper(plants);
    }

    // Recursive helper for remove_plant_from_pot
    fun remove_plant_from_pot_helper(plants_from: &mut vector<address>, plant_addr: address, i: u64): bool {
        if (i >= vector::length(plants_from)) {
            return false;
        };
        if (*vector::borrow(plants_from, i) == plant_addr) {
            vector::remove(plants_from, i);
            return true;
        };
        remove_plant_from_pot_helper(plants_from, plant_addr, i + 1)
    }

    fun remove_plant_from_pot(pot_addr: address, plant_addr: address) acquires Pot {
        assert!(exists<Pot>(pot_addr), 0x60007); // Pot does not exist
        let pot = borrow_global_mut<Pot>(pot_addr);
        let plants_from = &mut pot.current_plants;
        let found = remove_plant_from_pot_helper(plants_from, plant_addr, 0);
        assert!(found, 2007); // Error: Plant not found in source pot
    }

    fun add_plant_to_pot_and_update(pot_addr: address, plant_addr: address) acquires Pot, Plant {
        assert!(exists<Pot>(pot_addr), 0x60007); // Pot does not exist
        let pot = borrow_global_mut<Pot>(pot_addr);
        assert!(exists<Plant>(plant_addr), 0x60008); // Plant does not exist
        let plant = borrow_global_mut<Plant>(plant_addr);
        // Check if pot has available slot
        let current_count = vector::length(&pot.current_plants);
        assert!(current_count < (pot.slot_count as u64), 1001); // Error: Pot is full
        // Update plant's pot address and recalculate growth time
        plant.pot_address = option::some(pot_addr);
        let now = timestamp::now_seconds();
        let base_duration = plant.base_grow_duration_sec;
        let new_bonus = get_growth_bonus(pot.level);
        let elapsed_time = now - plant.planted_at;
        let remaining_base_time = base_duration - elapsed_time;
        let new_effective_remaining = remaining_base_time * (100 - new_bonus) / 100;
        plant.growth_complete_at = now + new_effective_remaining;
        plant.extended_growth_percent = new_bonus as u8;
        // Add plant to pot
        vector::push_back(&mut pot.current_plants, plant_addr);
    }

    public entry fun transfer_plant(pot_from_addr: address, pot_to_addr: address, plant_addr: address) acquires Pot, Plant {
        assert!(exists<Pot>(pot_from_addr), 0x60007); // Pot does not exist
        assert!(exists<Pot>(pot_to_addr), 0x60007); // Pot does not exist
        assert!(exists<Plant>(plant_addr), 0x60008); // Plant does not exist
        if (pot_from_addr == pot_to_addr) {
            assert!(exists<Pot>(pot_from_addr), 0x60007); // Pot does not exist
            let pot = borrow_global_mut<Pot>(pot_from_addr);
            assert!(exists<Plant>(plant_addr), 0x60008); // Plant does not exist
            let plant = borrow_global_mut<Plant>(plant_addr);
            // If transferring to the same pot, just update the plant's pot_address and growth time if needed
            plant.pot_address = option::some(pot_to_addr);
            let now = timestamp::now_seconds();
            let base_duration = plant.base_grow_duration_sec;
            let new_bonus = get_growth_bonus(pot.level);
            let elapsed_time = now - plant.planted_at;
            let remaining_base_time = base_duration - elapsed_time;
            let new_effective_remaining = remaining_base_time * (100 - new_bonus) / 100;
            plant.growth_complete_at = now + new_effective_remaining;
            plant.extended_growth_percent = new_bonus as u8;
        } else {
            remove_plant_from_pot(pot_from_addr, plant_addr);
            add_plant_to_pot_and_update(pot_to_addr, plant_addr);
        }
    }

    public entry fun apply_fertilizer(plant_addr: address, item_addr: address) acquires Plant, Item {
        assert!(exists<Plant>(plant_addr), 0x60008); // Plant does not exist
        let plant = borrow_global_mut<Plant>(plant_addr);
        assert!(exists<Item>(item_addr), 0x60009); // Item does not exist
        let item = borrow_global_mut<Item>(item_addr);
        
        assert!(item.usage_count < item.max_usage, 2004); // Error: Item usage limit reached
        assert!(item.usage_type == 0, 2005); // Error: Item is not fertilizer (usage_type = 0)
        assert!(plant.growth < 100, 2006); // Error: Plant growth already max
        
        // Apply fertilizer effect to growth (plant quality)
        plant.growth = plant.growth + (item.effect_value as u64);
        
        // Apply fertilizer effect to growth time (growth duration)
        let now = timestamp::now_seconds();
        let remaining_time = plant.growth_complete_at - now;
        let time_reduction = remaining_time * (item.effect_value as u64) / 100; // Reduce % of remaining time
        plant.growth_complete_at = plant.growth_complete_at - time_reduction;
        
        item.usage_count = item.usage_count + 1;
    }

    public entry fun advance_growth(plant_addr: address) acquires Plant, Pot {
        assert!(exists<Plant>(plant_addr), 0x60008); // Plant does not exist
        let plant = borrow_global_mut<Plant>(plant_addr);
        let base_growth: u64 = 10;
        let growth_bonus = if (option::is_some(&plant.pot_address)) {
            let pot_addr = *option::borrow(&plant.pot_address);
            assert!(exists<Pot>(pot_addr), 0x60007); // Pot does not exist
            let pot = borrow_global<Pot>(pot_addr);
            get_growth_bonus(pot.level)
        } else {
            0
        };
        let final_growth = base_growth + (base_growth * growth_bonus / 100);
        plant.growth = plant.growth + final_growth;
    }

    // Growth bonus by pot level
    fun get_growth_bonus(level: u8): u64 {
        if (level == 1) 5
        else if (level == 2) 15
        else if (level == 3) 30
        else if (level == 4) 40
        else if (level == 5) 50
        else 0
    }

    // Base growth duration is fixed (5 minutes = 300 seconds)
    fun get_base_growth_duration(): u64 {
        300
    }



    // ===================== Pet Functions =====================
    public entry fun evolve_to_pet(plant_addr: address, receiver: address) acquires Plant, Pot, PetCollectionOwnerConfig {
        assert!(exists<Plant>(plant_addr), 0x60008); // Plant does not exist
        let plant = borrow_global_mut<Plant>(plant_addr);
        assert!(plant.stage == 3, 2001); // Error: Plant not ready to evolve
        assert!(plant.is_ready_to_harvest, 2002); // Error: Plant not ready to harvest
        let pet_collection_address = get_pet_collection_owner_address();
        // Get pot level for rarity calculation
        let pot_level = if (option::is_some(&plant.pot_address)) {
            let pot_addr = *option::borrow(&plant.pot_address);
            assert!(exists<Pot>(pot_addr), 0x60007); // Pot does not exist
            let pot = borrow_global<Pot>(pot_addr);
            pot.level
        } else {
            1 // Default level if not planted in pot
        };
        // Burn the plant
        let Plant {
            plant_id,
            species,
            stage: _,
            growth,
            planted_at: _,
            base_grow_duration_sec: _,
            extended_growth_percent: _,
            growth_complete_at: _,
            is_ready_to_harvest: _,
            pot_address,
            quantity: _,
            extend_ref: _,
            burn_ref,
        } = move_from<Plant>(plant_addr);
        token::burn(burn_ref);
        // Remove from pot if planted
        if (option::is_some(&pot_address)) {
            let pot_addr = *option::borrow(&pot_address);
            assert!(exists<Pot>(pot_addr), 0x60007); // Pot does not exist
            let pot = borrow_global_mut<Pot>(pot_addr);
            let plants = &mut pot.current_plants;
            remove_plant_from_pot_in_evolve(plants, plant_addr, 0);
        };
        // Generate pet stats based on plant_id, growth and pot level
        let (intelligence, strength, agility) = get_pet_stats(plant_id, growth, pot_level);
        // Create pet NFT
        let constructor_ref = &token::create(
            &get_pet_collection_signer(pet_collection_address),
            string::utf8(PET_COLLECTION_NAME),
            string::utf8(PET_COLLECTION_DESCRIPTION),
            species,
            option::none(),
            string::utf8(PET_COLLECTION_URI),
        );
        let token_signer_ref = &object::generate_signer(constructor_ref);
        let pet_address = signer::address_of(token_signer_ref);
        let extend_ref = object::generate_extend_ref(constructor_ref);
        let burn_ref = token::generate_burn_ref(constructor_ref);
        let transfer_ref = object::generate_transfer_ref(constructor_ref);
        let pet = Pet {
            name: species,
            species: species,
            intelligence,
            strength,
            agility,
            evolved_from_plant: plant_addr,
            extend_ref,
            burn_ref,
        };
        move_to(token_signer_ref, pet);
        // Emit events
        event::emit<MintPetEvent>(MintPetEvent {
            pet_address,
            name: species,
            species: species,
            intelligence,
            strength,
            agility,
            evolved_from_plant: plant_addr,
            receiver,
        });
        event::emit<EvolvePlantToPetEvent>(EvolvePlantToPetEvent {
            plant_address: plant_addr,
            pet_address,
            species: species,
            receiver,
        });
        // Transfer pet to receiver
        object::transfer_with_ref(object::generate_linear_transfer_ref(&transfer_ref), receiver);
    }

    fun get_pet_stats(plant_id: u8, growth: u64, pot_level: u8): (u8, u8, u8) {
        // Base stats depend on plant_id (species characteristics)
        let (species_int, species_str, species_agi) = if (plant_id == 1) {
            (60, 50, 90) // Carrot: High agility, medium intelligence, low strength
        } else if (plant_id == 2) {
            (90, 70, 60) // Tomato: High intelligence, medium strength, low agility
        } else if (plant_id == 3) {
            (70, 90, 50) // Wheat: High strength, medium intelligence, low agility
        } else if (plant_id == 4) {
            (80, 80, 70) // Corn: Balanced high stats
        } else if (plant_id == 5) {
            (95, 60, 80) // Rose: Very high intelligence, good agility, medium strength
        } else {
            (60, 60, 60) // Default balanced stats
        };
        
        // Growth multiplier (0.5x to 1.5x based on growth)
        let growth_multiplier = if (growth >= 80) 150        // 1.5x for excellent growth
        else if (growth >= 60) 125                           // 1.25x for good growth
        else if (growth >= 40) 100                           // 1.0x for average growth
        else if (growth >= 20) 75                            // 0.75x for poor growth
        else 50;                                             // 0.5x for very poor growth
        
        // Apply growth multiplier
        let growth_int = (species_int as u64) * growth_multiplier / 100;
        let growth_str = (species_str as u64) * growth_multiplier / 100;
        let growth_agi = (species_agi as u64) * growth_multiplier / 100;
        
        // Apply pot level bonus
        let level_bonus = (pot_level - 1) * 5; // Level 1: 0%, Level 2: 5%, Level 3: 10%, etc.
        
        let final_int = if (growth_int + (level_bonus as u64) > 100) 100 else (growth_int + (level_bonus as u64)) as u8;
        let final_str = if (growth_str + (level_bonus as u64) > 100) 100 else (growth_str + (level_bonus as u64)) as u8;
        let final_agi = if (growth_agi + (level_bonus as u64) > 100) 100 else (growth_agi + (level_bonus as u64)) as u8;
        
        (final_int, final_str, final_agi)
    }

    // Recursive helper for removing plant from pot in evolve_to_pet
    fun remove_plant_from_pot_in_evolve(plants: &mut vector<address>, plant_addr: address, i: u64) {
        if (i >= vector::length(plants)) {
            return;
        };
        if (*vector::borrow(plants, i) == plant_addr) {
            vector::remove(plants, i);
            return;
        };
        remove_plant_from_pot_in_evolve(plants, plant_addr, i + 1)
    }

    // Create Item with custom parameters
    public fun create_item_with_params(
        user: &signer,
        name: String,
        description: String,
        uri: String,
        usage_type: u8,
        effect_value: u8,
        max_usage: u8,
        receive_addr: address
    ): address acquires ItemCollectionOwnerConfig {
        let item_collection_address = get_item_collection_owner_address();
        let constructor_ref = &token::create(
            &get_item_collection_signer(item_collection_address),
            name,
            description,
            name,
            option::none(),
            uri,
        );
        let token_signer_ref = &object::generate_signer(constructor_ref);
        let item_address = signer::address_of(token_signer_ref);

        let extend_ref = object::generate_extend_ref(constructor_ref);
        let burn_ref = token::generate_burn_ref(constructor_ref);
        let transfer_ref = object::generate_transfer_ref(constructor_ref);

        let item = Item {
            name: name,
            description: description,
            usage_type,
            effect_value,
            usage_count: 0,
            max_usage,
            quantity: 1,
            extend_ref,
            burn_ref,
        };
        move_to(token_signer_ref, item);

        // Emit event for minting Item
        event::emit<MintItemEvent>(MintItemEvent {
            item_address,
            name,
            description,
            uri,
            receiver: receive_addr,
        });

        // Transfer the Item to the receive_addr
        object::transfer_with_ref(object::generate_linear_transfer_ref(&transfer_ref), receive_addr);

        item_address
    }
